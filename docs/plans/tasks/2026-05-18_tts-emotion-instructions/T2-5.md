```typescript
import { describe, it, expect } from 'vitest';
import {
  TTS_EMOTION_PRESETS,
  type TTSEmotionPreset,
} from './tts-emotion-presets';

describe('TTS_EMOTION_PRESETS', () => {
  it('should have exactly 6 presets', () => {
    expect(TTS_EMOTION_PRESETS).toHaveLength(6);
  });

  it('should contain all expected emotion keys', () => {
    const keys = TTS_EMOTION_PRESETS.map((p) => p.key);
    expect(keys).toContain('joy');
    expect(keys).toContain('sadness');
    expect(keys).toContain('anger');
    expect(keys).toContain('surprise');
    expect(keys).toContain('calm');
    expect(keys).toContain('confusion');
  });

  it('each preset should have non-empty English instructions', () => {
    for (const preset of TTS_EMOTION_PRESETS) {
      expect(preset.instructions).toBeTruthy();
      // instructions は英語固定 (ASCII 文字が主体)
      expect(/[a-zA-Z]/.test(preset.instructions)).toBe(true);
    }
  });

  it('each preset should have an emoji', () => {
    for (const preset of TTS_EMOTION_PRESETS) {
      expect(preset.emoji).toBeTruthy();
    }
  });

  it('each preset should have a Japanese label', () => {
    for (const preset of TTS_EMOTION_PRESETS) {
      expect(preset.labelJa).toBeTruthy();
    }
  });
});
```

### Step 3: DialogueNode テストを作成

`components/node-editor/nodes/DialogueNode.test.tsx`:

注意: `@xyflow/react` の Context が必要なため `<ReactFlowProvider>` でラップする。

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { render, screen, fireEvent } from '@testing-library/react';
import { ReactFlowProvider } from '@xyflow/react';
import { DialogueNode } from './DialogueNode';
import type { DialogueNodeData } from '@/lib/types/node-editor';

// ttsApi をモック
vi.mock('@/lib/api/client', () => ({
  ttsApi: {
    listVoices: vi.fn().mockResolvedValue([]),
  },
  dialogueApi: {
    create: vi.fn(),
    getStatus: vi.fn(),
  },
}));

// window.dispatchEvent の nodeDataUpdate をキャプチャ
const captureNodeDataUpdate = () => {
  const updates: Record<string, unknown>[] = [];
  const handler = (e: Event) => {
    updates.push((e as CustomEvent).detail.updates);
  };
  window.addEventListener('nodeDataUpdate', handler);
  return {
    updates,
    cleanup: () => window.removeEventListener('nodeDataUpdate', handler),
  };
};

const makeDialogueData = (overrides?: Partial<DialogueNodeData>): DialogueNodeData => ({
  type: 'dialogue',
  isValid: false,
  text: 'テストセリフ',
  voiceId: 'alloy',
  language: 'ja',
  speed: 1.0,
  useLipSync: false,
  ttsInstructions: undefined,
  status: 'idle',
  progress: 0,
  generationId: null,
  outputVideoUrl: null,
  ...overrides,
});

const renderDialogueNode = (data: DialogueNodeData) => {
  return render(
    <ReactFlowProvider>
      <DialogueNode
        id="node-test-001"
        data={data}
        selected={false}
        type="dialogue"
        dragging={false}
        zIndex={0}
        isConnectable={true}
        positionAbsoluteX={0}
        positionAbsoluteY={0}
      />
    </ReactFlowProvider>
  );
};

describe('DialogueNode 感情パネル', () => {
  it('初期表示で感情パネルが折りたたまれていること (AC8)', () => {
    renderDialogueNode(makeDialogueData());
    // ヘッダは可視
    expect(screen.getByText('感情・トーン (任意)')).toBeInTheDocument();
    // プリセットボタンは非表示
    expect(screen.queryByText('喜び')).not.toBeInTheDocument();
  });

  it('ヘッダクリックで感情パネルが展開される', () => {
    renderDialogueNode(makeDialogueData());
    fireEvent.click(screen.getByText('感情・トーン (任意)'));
    // 展開後はプリセットボタンが表示される
    expect(screen.getByText('喜び')).toBeInTheDocument();
    expect(screen.getByText('悲しみ')).toBeInTheDocument();
    expect(screen.getByText('怒り')).toBeInTheDocument();
    expect(screen.getByText('驚き')).toBeInTheDocument();
    expect(screen.getByText('落ち着き')).toBeInTheDocument();
    expect(screen.getByText('困惑')).toBeInTheDocument();
  });

  it('展開後に注記が常時表示される (A 案 / AC7)', () => {
    renderDialogueNode(makeDialogueData());
    fireEvent.click(screen.getByText('感情・トーン (任意)'));
    expect(
      screen.getByText(/instructions は OpenAI TTS のみで有効/)
    ).toBeInTheDocument();
  });

  it('プリセット「喜び」押下で ttsInstructions がセットされる (AC2)', () => {
    const { updates, cleanup } = captureNodeDataUpdate();
    renderDialogueNode(makeDialogueData());
    fireEvent.click(screen.getByText('感情・トーン (任意)'));
    fireEvent.click(screen.getByText('喜び'));

    expect(updates.length).toBeGreaterThan(0);
    const lastUpdate = updates[updates.length - 1];
    expect(lastUpdate).toHaveProperty('ttsInstructions');
    expect((lastUpdate.ttsInstructions as string)).toContain('cheerful enthusiasm');
    cleanup();
  });

  it('textarea 編集で ttsInstructions が更新される (AC4)', () => {
    const { updates, cleanup } = captureNodeDataUpdate();
    renderDialogueNode(makeDialogueData());
    fireEvent.click(screen.getByText('感情・トーン (任意)'));

    const textarea = screen.getByPlaceholderText(/Whisper softly/);
    fireEvent.change(textarea, { target: { value: 'Speak dramatically' } });

    const textareaUpdates = updates.filter(
      (u) => u.ttsInstructions === 'Speak dramatically'
    );
    expect(textareaUpdates.length).toBeGreaterThan(0);
    cleanup();
  });

  it('textarea を空にすると ttsInstructions が undefined になる (AC10a)', () => {
    const { updates, cleanup } = captureNodeDataUpdate();
    renderDialogueNode(makeDialogueData({ ttsInstructions: 'existing text' }));
    fireEvent.click(screen.getByText('感情・トーン (任意)'));

    const textarea = screen.getByPlaceholderText(/Whisper softly/);
    fireEvent.change(textarea, { target: { value: '' } });

    const emptyUpdates = updates.filter((u) => u.ttsInstructions === undefined);
    expect(emptyUpdates.length).toBeGreaterThan(0);
    cleanup();
  });

  it('クリアボタンで ttsInstructions が undefined に戻る', () => {
    const { updates, cleanup } = captureNodeDataUpdate();
    renderDialogueNode(makeDialogueData({ ttsInstructions: 'custom instructions' }));
    fireEvent.click(screen.getByText('感情・トーン (任意)'));

    // クリアボタンをクリック (title 属性でセレクト)
    const clearBtn = screen.getByTitle('クリア (デフォルト適用に戻す)');
    fireEvent.click(clearBtn);

    const clearUpdates = updates.filter((u) => u.ttsInstructions === undefined);
    expect(clearUpdates.length).toBeGreaterThan(0);
    cleanup();
  });

  it('createDefaultNodeData の dialogue で ttsInstructions が undefined', () => {
    // node-editor.ts の createDefaultNodeData('dialogue') のテスト
    const { createDefaultNodeData } = require('@/lib/types/node-editor');
    const data = createDefaultNodeData('dialogue');
    expect(data.ttsInstructions).toBeUndefined();
  });
});
```
