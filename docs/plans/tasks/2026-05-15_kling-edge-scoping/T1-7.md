```typescript
import { test, expect } from '@playwright/test';

test.describe('KlingEdgeScoping - Handle 接続ガード', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/generate');
    // NodeEditor が表示されるのを待つ
    await page.waitForSelector('[data-testid="node-editor"]', { timeout: 10000 });
  });

  test('ProviderNode に 6 個の target Handle が表示される', async ({ page }) => {
    // ProviderNode をキャンバスに配置 (ドラッグ or ダブルクリックで追加)
    // 実際の追加操作は既存 E2E テストのパターンに従う
    // Handle の data-handleid 属性で 6 個を確認
    const handleIds = [
      'kling_mode_input',
      'kling_elements_input',
      'kling_end_frame_input',
      'kling_camera_control_input',
      'act_two_input',
      'hailuo_end_frame_input',
    ];
    for (const id of handleIds) {
      await expect(page.locator(`[data-handleid="${id}"]`).first()).toBeVisible();
    }
  });
});
```
