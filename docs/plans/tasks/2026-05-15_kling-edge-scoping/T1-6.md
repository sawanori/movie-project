#### ケース C: エッジ接続なし (フォールバック) - console.warn と window.va が呼ばれる

```typescript
it('KlingElementsNode が未接続の場合、フォールバック発火 + console.warn + window.va が呼ばれる', () => {
  const providerNodeId = crypto.randomUUID();
  const generateNodeId = crypto.randomUUID();
  const elemId = crypto.randomUUID();

  // window.va をモック
  const vaSpy = vi.fn();
  Object.defineProperty(window, 'va', { value: vaSpy, writable: true });
  const warnSpy = vi.spyOn(console, 'warn').mockImplementation(() => {});

  const nodes = [
    createImageInputNode({ imageUrl: 'https://example.com/img.jpg' }),
    createPromptNode({ englishPrompt: 'test' }),
    createProviderNode({ id: providerNodeId, provider: 'piapi_kling' }),
    createKlingElementsNode({ id: elemId, elementImages: ['https://example.com/e1.jpg'] }),
    createGenerateNode({ id: generateNodeId }),
  ];
  // ProviderNode は GenerateNode に接続されているが、KlingElementsNode は未接続
  const edges = [
    createEdge({ source: providerNodeId, target: generateNodeId, targetHandle: HANDLE_IDS.CONFIG_INPUT }),
  ];

  const request = graphToStoryVideoCreate(nodes, edges, generateNodeId);
  // フォールバックで element_images は設定される
  expect(request.element_images).toEqual(['https://example.com/e1.jpg']);
  // 警告ログが出る
  expect(warnSpy).toHaveBeenCalledWith(expect.stringContaining('klingElements'));
  // Telemetry event が送信される
  expect(vaSpy).toHaveBeenCalledWith('event', expect.objectContaining({
    name: 'kling_edge_scoping_fallback',
  }));

  warnSpy.mockRestore();
});
```

### Step 3: useWorkflowValidation.test.ts を新規作成

```typescript
import { renderHook } from '@testing-library/react';
import { useWorkflowValidation } from './useWorkflowValidation';
import { HANDLE_IDS } from '@/lib/types/node-editor';

// テスト用ノード/エッジ生成ヘルパー
// ... (ファクトリ関数をここに定義)

describe('useWorkflowValidation', () => {
  it('ActTwoNode が piapi_kling Provider に接続されている場合、provider_mismatch error が発生する', () => {
    const provId = crypto.randomUUID();
    const actTwoId = crypto.randomUUID();
    const genId = crypto.randomUUID();
    const nodes = [
      createProviderNode({ id: provId, provider: 'piapi_kling' }),
      createActTwoNode({ id: actTwoId, useActTwo: true }),
      createGenerateNode({ id: genId }),
      createImageInputNode({ imageUrl: 'https://example.com/img.jpg' }),
      createPromptNode({ englishPrompt: 'test' }),
    ];
    const edges = [
      createEdge({ source: provId, target: genId, targetHandle: HANDLE_IDS.CONFIG_INPUT }),
      createEdge({ source: actTwoId, target: provId, targetHandle: HANDLE_IDS.ACT_TWO_INPUT }),
    ];
    const { result } = renderHook(() => useWorkflowValidation(nodes, edges));
    expect(result.current.errors.some(e => e.type === 'provider_mismatch')).toBe(true);
  });

  it('設定ノードが未接続の場合、disconnected_optional warning が発生する', () => {
    const nodes = [
      createProviderNode({ provider: 'piapi_kling' }),
      createKlingElementsNode({ elementImages: [] }),
      createGenerateNode(),
      createImageInputNode({ imageUrl: 'https://example.com/img.jpg' }),
      createPromptNode({ englishPrompt: 'test' }),
    ];
    const { result } = renderHook(() => useWorkflowValidation(nodes, []));
    expect(result.current.warnings.some(w => w.type === 'disconnected_optional')).toBe(true);
  });

  it('複数 Provider + CONFIG_INPUT 未接続の GenerateNode で ambiguous_provider warning が発生する', () => {
    const nodes = [
      createProviderNode({ provider: 'piapi_kling' }),
      createProviderNode({ provider: 'runway' }),
      createGenerateNode(),  // CONFIG_INPUT 未接続
      createImageInputNode({ imageUrl: 'https://example.com/img.jpg' }),
      createPromptNode({ englishPrompt: 'test' }),
    ];
    const { result } = renderHook(() => useWorkflowValidation(nodes, []));
    expect(result.current.warnings.some(w => w.type === 'ambiguous_provider')).toBe(true);
  });

  it('全エッジ正常 (複数 Generate + 複数 Provider) で errors / warnings が空', () => {
    const provA = crypto.randomUUID();
    const genA = crypto.randomUUID();
    const nodes = [
      createProviderNode({ id: provA, provider: 'piapi_kling' }),
      createGenerateNode({ id: genA }),
      createImageInputNode({ imageUrl: 'https://example.com/img.jpg' }),
      createPromptNode({ englishPrompt: 'test' }),
    ];
    const edges = [
      createEdge({ source: provA, target: genA, targetHandle: HANDLE_IDS.CONFIG_INPUT }),
    ];
    const { result } = renderHook(() => useWorkflowValidation(nodes, edges));
    expect(result.current.errors).toHaveLength(0);
    expect(result.current.warnings).toHaveLength(0);
  });

  it('CONFIG_INPUT に複数 Provider 接続で multiple_provider_connection error が発生する', () => {
    const provA = crypto.randomUUID();
    const provB = crypto.randomUUID();
    const genId = crypto.randomUUID();
    const nodes = [
      createProviderNode({ id: provA, provider: 'piapi_kling' }),
      createProviderNode({ id: provB, provider: 'runway' }),
      createGenerateNode({ id: genId }),
      createImageInputNode({ imageUrl: 'https://example.com/img.jpg' }),
      createPromptNode({ englishPrompt: 'test' }),
    ];
    const edges = [
      createEdge({ source: provA, target: genId, targetHandle: HANDLE_IDS.CONFIG_INPUT }),
      createEdge({ source: provB, target: genId, targetHandle: HANDLE_IDS.CONFIG_INPUT }),
    ];
    const { result } = renderHook(() => useWorkflowValidation(nodes, edges));
    expect(result.current.errors.some(e => e.type === 'multiple_provider_connection')).toBe(true);
  });
});
```

### Step 4: ProviderNode.test.tsx を新規作成

```typescript
import { render, screen } from '@testing-library/react';
import { ReactFlowProvider } from '@xyflow/react';
import { ProviderNode } from './ProviderNode';
import { HANDLE_IDS } from '@/lib/types/node-editor';

describe('ProviderNode', () => {
  const baseData = {
    type: 'provider' as const,
    provider: 'piapi_kling' as const,
    // ... 必要な最小フィールド
  };

  it('6 個の target Handle が全てレンダリングされる', () => {
    render(
      <ReactFlowProvider>
        <ProviderNode id="test-node" data={baseData} selected={false} />
      </ReactFlowProvider>
    );
    // xyflow の Handle は data-handleid 属性を持つ
    const expectedIds = [
      HANDLE_IDS.KLING_MODE_INPUT,
      HANDLE_IDS.KLING_ELEMENTS_INPUT,
      HANDLE_IDS.KLING_END_FRAME_INPUT,
      HANDLE_IDS.KLING_CAMERA_CONTROL_INPUT,
      HANDLE_IDS.ACT_TWO_INPUT,
      HANDLE_IDS.HAILUO_END_FRAME_INPUT,
    ];
    for (const id of expectedIds) {
      expect(
        document.querySelector(`[data-handleid="${id}"]`)
      ).toBeInTheDocument();
    }
  });

  it('provider=runway の場合、KLING_* Handle が opacity-30 クラスを持つ', () => {
    const runwayData = { ...baseData, provider: 'runway' as const };
    render(
      <ReactFlowProvider>
        <ProviderNode id="test-node" data={runwayData} selected={false} />
      </ReactFlowProvider>
    );
    const klingHandle = document.querySelector(
      `[data-handleid="${HANDLE_IDS.KLING_MODE_INPUT}"]`
    );
    expect(klingHandle).toHaveClass('opacity-30');

    const actTwoHandle = document.querySelector(
      `[data-handleid="${HANDLE_IDS.ACT_TWO_INPUT}"]`
    );
    expect(actTwoHandle).not.toHaveClass('opacity-30');
  });
});
```
