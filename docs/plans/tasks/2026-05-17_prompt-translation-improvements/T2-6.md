```tsx
/**
 * PromptNode.test.tsx
 * AC-C4 確認カード表示 / C3 セリフなし非表示 / N3 dismiss 抑制 / N4 CustomEvent 発火 / B2 subject_type 送信
 */
import { render, screen, waitFor, fireEvent } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { vi, describe, it, expect, beforeEach } from 'vitest';
import { ReactFlowProvider } from '@xyflow/react';
import { PromptNode } from '../PromptNode';
import { videosApi } from '@/lib/api/client';

// ReactFlow provider でラップするヘルパー
function renderWithFlow(ui: React.ReactElement) {
  return render(<ReactFlowProvider>{ui}</ReactFlowProvider>);
}

// videosApi.translateStoryPrompt を mock
vi.mock('@/lib/api/client', () => ({
  videosApi: {
    translateStoryPrompt: vi.fn(),
  },
}));

const mockTranslate = videosApi.translateStoryPrompt as ReturnType<typeof vi.fn>;

const defaultData = {
  japanesePrompt: '',
  englishPrompt: '',
  isValid: false,
  isTranslating: false,
  subjectType: 'person' as const,
};

describe('PromptNode', () => {
  beforeEach(() => {
    vi.clearAllMocks();
    // デフォルト: extracted_dialogue なし
    mockTranslate.mockResolvedValue({
      english_prompt: 'Preserve identity. CLIP SPECIFIC: Action: wave.',
      extracted_dialogue: null,
    });
  });

  it('does not show dialogue card when extracted_dialogue is null (AC-C3)', async () => {
    renderWithFlow(
      <PromptNode data={defaultData} selected={false} id="node-1" />
    );
    const textarea = screen.getByPlaceholderText('動画の内容を日本語で入力');
    await userEvent.type(textarea, '女性が振り向く');

    await waitFor(() => {
      expect(screen.queryByText('セリフを検出しました')).not.toBeInTheDocument();
    }, { timeout: 2000 });
  });

  it('shows dialogue card when extracted_dialogue is detected (AC-C4)', async () => {
    mockTranslate.mockResolvedValue({
      english_prompt: 'Preserve design. ...',
      extracted_dialogue: 'ちょっとまって…',
    });

    renderWithFlow(
      <PromptNode data={defaultData} selected={false} id="node-1" />
    );
    const textarea = screen.getByPlaceholderText('動画の内容を日本語で入力');
    await userEvent.type(textarea, '「ちょっとまって…」と言う');

    await waitFor(() => {
      expect(screen.getByText('セリフを検出しました')).toBeInTheDocument();
      expect(screen.getByText(/ちょっとまって/)).toBeInTheDocument();
    }, { timeout: 2000 });
  });

  it('dispatches createDialogueNodeFromPrompt event on button click (N4, AC-C5)', async () => {
    mockTranslate.mockResolvedValue({
      english_prompt: 'Preserve design.',
      extracted_dialogue: 'ちょっとまって…',
    });

    const spy = vi.spyOn(window, 'dispatchEvent');

    renderWithFlow(
      <PromptNode data={defaultData} selected={false} id="node-1" />
    );
    const textarea = screen.getByPlaceholderText('動画の内容を日本語で入力');
    await userEvent.type(textarea, 'セリフ入力');

    await waitFor(() => {
      expect(screen.getByText('新規 DialogueNode を作成')).toBeInTheDocument();
    }, { timeout: 2000 });

    await userEvent.click(screen.getByText('新規 DialogueNode を作成'));

    expect(spy).toHaveBeenCalledWith(
      expect.objectContaining({
        type: 'createDialogueNodeFromPrompt',
        detail: expect.objectContaining({ initialText: 'ちょっとまって…' }),
      })
    );
    spy.mockRestore();
  });

  it('hides card after dismiss and does not re-show for same dialogue (N3)', async () => {
    mockTranslate.mockResolvedValue({
      english_prompt: 'Preserve design.',
      extracted_dialogue: 'ちょっとまって…',
    });

    renderWithFlow(
      <PromptNode data={defaultData} selected={false} id="node-1" />
    );
    const textarea = screen.getByPlaceholderText('動画の内容を日本語で入力');
    await userEvent.type(textarea, '入力1');

    await waitFor(() => {
      expect(screen.getByText('セリフを検出しました')).toBeInTheDocument();
    }, { timeout: 2000 });

    await userEvent.click(screen.getByText('無視'));
    expect(screen.queryByText('セリフを検出しました')).not.toBeInTheDocument();

    // 同じ (正規化後同一の) セリフで再翻訳 → カード非表示のまま
    await userEvent.clear(textarea);
    await userEvent.type(textarea, '入力2');

    await waitFor(() => {
      // extracted_dialogue が同じなので再表示されない
      expect(screen.queryByText('セリフを検出しました')).not.toBeInTheDocument();
    }, { timeout: 2000 });
  });

  it('passes subject_type from node data to translateStoryPrompt (B2)', async () => {
    const objectData = { ...defaultData, subjectType: 'object' as const };

    renderWithFlow(
      <PromptNode data={objectData} selected={false} id="node-1" />
    );
    const textarea = screen.getByPlaceholderText('動画の内容を日本語で入力');
    await userEvent.type(textarea, 'ニットセーター');

    await waitFor(() => {
      expect(mockTranslate).toHaveBeenCalledWith(
        expect.objectContaining({ subject_type: 'object' })
      );
    }, { timeout: 2000 });
  });
});
```

```tsx
/**
 * NodeEditor.dialogue.test.tsx
 * createDialogueNodeFromPrompt listener 動作確認 (AC-C5)
 */
import { render, waitFor } from '@testing-library/react';
import { vi, describe, it, expect } from 'vitest';
import userEvent from '@testing-library/user-event';

// NodeEditor は重いので、CustomEvent listener 部分のみテスト
// setNodes の関数形式 setter が呼ばれることを検証する

describe('NodeEditor createDialogueNodeFromPrompt listener', () => {
  it('adds a dialogue node when event is dispatched (AC-C5)', async () => {
    // CustomEvent を手動 dispatch してリスナーの動作を確認
    const receivedEvents: CustomEvent[] = [];
    const handler = (e: Event) => receivedEvents.push(e as CustomEvent);
    window.addEventListener('createDialogueNodeFromPrompt', handler);

    const event = new CustomEvent('createDialogueNodeFromPrompt', {
      detail: {
        sourcePromptNodeId: 'prompt-1',
        initialText: 'ちょっとまって…',
      },
    });
    window.dispatchEvent(event);

    expect(receivedEvents).toHaveLength(1);
    expect(receivedEvents[0].detail.initialText).toBe('ちょっとまって…');

    window.removeEventListener('createDialogueNodeFromPrompt', handler);
  });
});
```
