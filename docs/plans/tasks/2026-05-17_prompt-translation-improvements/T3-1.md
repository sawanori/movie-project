```tsx
/**
 * NodeEditor.integration.test.tsx
 *
 * FE → API → PromptNode 確認カード → DialogueNode 配置の E2E ライクな統合テスト。
 * API 呼び出しは vi.mock で mock。NodeEditor 全体をレンダリングして動作を確認する。
 *
 * 注意: NodeEditor は ReactFlow を大量に使うため、@xyflow/react の mock が必要な場合がある。
 * 代替として PromptNode + CustomEvent ベースのテストを採用する。
 */
import { render, screen, waitFor, act } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { vi, describe, it, expect, beforeEach } from 'vitest';
import { ReactFlowProvider } from '@xyflow/react';
import { videosApi } from '@/lib/api/client';

vi.mock('@/lib/api/client', () => ({
  videosApi: {
    translateStoryPrompt: vi.fn(),
  },
  dialogueApi: {},
  utilityApi: {},
}));

const mockTranslate = videosApi.translateStoryPrompt as ReturnType<typeof vi.fn>;

/**
 * 統合テスト 1:
 * PromptNode でセリフ入力 → 確認カード表示 → 「新規作成」クリック →
 * createDialogueNodeFromPrompt CustomEvent が発火する
 */
describe('PromptNode → DialogueNode integration', () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it('dialogue card appears and fires createDialogueNodeFromPrompt on button click', async () => {
    mockTranslate.mockResolvedValue({
      english_prompt: 'Preserve object design. CLIP SPECIFIC: ...',
      extracted_dialogue: 'ちょっとまって…',
    });

    // PromptNode 単体を ReactFlowProvider でラップしてテスト
    // (NodeEditor 全体は重いため、CustomEvent ベースで統合を検証)
    const { PromptNode } = await import('../nodes/PromptNode');

    const receivedEvents: CustomEvent[] = [];
    const eventSpy = (e: Event) => receivedEvents.push(e as CustomEvent);
    window.addEventListener('createDialogueNodeFromPrompt', eventSpy);

    const defaultData = {
      japanesePrompt: '',
      englishPrompt: '',
      isValid: false,
      isTranslating: false,
      subjectType: 'object' as const,
    };

    render(
      <ReactFlowProvider>
        <PromptNode data={defaultData} selected={false} id="prompt-integration-1" />
      </ReactFlowProvider>
    );

    const textarea = screen.getByPlaceholderText('動画の内容を日本語で入力');
    await userEvent.type(textarea, 'ニットセーターが「ちょっとまって…」と言う');

    // 確認カード表示を待つ
    await waitFor(() => {
      expect(screen.getByText('セリフを検出しました')).toBeInTheDocument();
    }, { timeout: 2000 });

    // 「新規 DialogueNode を作成」をクリック
    await userEvent.click(screen.getByText('新規 DialogueNode を作成'));

    // CustomEvent が発火されていること
    expect(receivedEvents).toHaveLength(1);
    expect(receivedEvents[0].detail).toEqual(
      expect.objectContaining({
        sourcePromptNodeId: 'prompt-integration-1',
        initialText: 'ちょっとまって…',
      })
    );

    // カードが非表示になること
    expect(screen.queryByText('セリフを検出しました')).not.toBeInTheDocument();

    window.removeEventListener('createDialogueNodeFromPrompt', eventSpy);
  });

  it('subject_type=object is sent to translateStoryPrompt (B2 integration)', async () => {
    mockTranslate.mockResolvedValue({
      english_prompt: 'Preserve object design.',
      extracted_dialogue: null,
    });

    const { PromptNode } = await import('../nodes/PromptNode');

    const objectData = {
      japanesePrompt: '',
      englishPrompt: '',
      isValid: false,
      isTranslating: false,
      subjectType: 'object' as const,
    };

    render(
      <ReactFlowProvider>
        <PromptNode data={objectData} selected={false} id="prompt-b2" />
      </ReactFlowProvider>
    );

    const textarea = screen.getByPlaceholderText('動画の内容を日本語で入力');
    await userEvent.type(textarea, 'オブジェクト入力');

    await waitFor(() => {
      expect(mockTranslate).toHaveBeenCalledWith(
        expect.objectContaining({ subject_type: 'object' })
      );
    }, { timeout: 2000 });

    // english_prompt に人物フレーズが含まれていないことを確認
    // (BE は mock なので BE の実際の出力ではなく、リクエストの correctness を確認)
    expect(mockTranslate).not.toHaveBeenCalledWith(
      expect.objectContaining({ subject_type: 'person' })
    );
  });
});
```
